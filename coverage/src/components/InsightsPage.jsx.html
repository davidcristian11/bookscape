import React, { useMemo, useState, useRef, useEffect } from 'react';
import { PieChart, Pie, Cell, Tooltip as RechartsTooltip, ResponsiveContainer, BarChart, Bar, XAxis, YAxis, CartesianGrid } from 'recharts';
import { faker } from '@faker-js/faker';
import './Insights.css';
 
export default function InsightsPage({ books, onAdd }) {
    // ==========================================
    // LOGICA PENTRU AUTO-GENERARE (STRESS TEST)
    // ==========================================
    const [isGenerating, setIsGenerating] = useState(false);
 
    // Folosim useRef pentru a stoca ID-urile cronometrelor fără a declanșa re-randări inutile
    const timeoutRef = useRef(null);
    const intervalRef = useRef(null);
 
    // Funcția care creează o carte falsă, dar realistă
    const generateFakeBook = () => {
        return {
            id: Date.now() + Math.random(),
            title: faker.lorem.words({ min: 2, max: 5 }),
            author: faker.person.fullName(),
            genre: faker.helpers.arrayElement(['Fiction', 'Memoir', 'Historical Fiction', 'Science Fiction', 'Self-Help']),
            rating: faker.number.int({ min: 1, max: 5 }),
            source: faker.helpers.arrayElement(['Amazon', 'Goodreads', 'Barnes & Noble']),
            month: faker.helpers.arrayElement(['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun'])
        };
    };
 
    const toggleAutoGenerate = () => {
        if (isGenerating) {
            // 1. DACĂ E PORNIT -> ÎL OPRIM
            setIsGenerating(false);
            clearTimeout(timeoutRef.current);
            clearInterval(intervalRef.current);
        } else {
            // 2. DACĂ E OPRIT -> ÎL PORNIM
            setIsGenerating(true);
 
            // Așteptăm 1.5 secunde (1500 ms) înainte să înceapă nebunia
            timeoutRef.current = setTimeout(() => {
 
                // După cele 1.5 secunde, pornim un interval care adaugă o carte la fiecare 500ms (jumătate de secundă)
                intervalRef.current = setInterval(() => {
                    onAdd(generateFakeBook());
                }, 500);
 
            }, 1500);
        }
    };
 
    // MĂSURĂ DE SIGURANȚĂ: Dacă utilizatorul schimbă pagina pe "Library" în timp ce
    // butonul rulează, trebuie să curățăm intervalul, altfel va genera cărți la infinit în fundal!
    useEffect(() => {
        return () => {
            clearTimeout(timeoutRef.current);
            clearInterval(intervalRef.current);
        };
    }, []);
 
    // ==========================================
    // 1. Calcularea datelor pentru "Library by Genre"
    // ==========================================
    // Folosim useMemo pentru a recalcula aceste date DOAR când lista de cărți se schimbă
    const genreData = useMemo(() => {
        // Reducem lista de cărți la un dicționar cu genurile și numărul lor
        const counts = books.reduce((acc, book) => {
            acc[book.genre] = (acc[book.genre] || 0) + 1;
            return acc;
        }, {});
 
        // Transformăm dicționarul în formatul cerut de Recharts
        return Object.keys(counts).map(genre => ({
            name: genre,
            value: counts[genre]
        }));
    }, [books]);
 
    // Culorile din designul tău pentru graficul Donut
    const COLORS = ['#2C4A3E', '#C75B33', '#1E332A', '#8c8c8c', '#5a7d6d'];
 
    // ==========================================
    // 2. Calcularea Top Surse (Top Rated Sources)
    // ==========================================
    const topSources = useMemo(() => {
        // Grupăm cărțile după sursă și le calculăm media
        const sourceStats = books.reduce((acc, book) => {
            if (!acc[book.source]) {
                acc[book.source] = { totalRating: 0, count: 0 };
            }
            acc[book.source].totalRating += book.rating;
            acc[book.source].count += 1;
            return acc;
        }, {});
 
        // Transformăm și sortăm descrescător după medie
        return Object.keys(sourceStats)
            .map(source => ({
                name: source,
                average: (sourceStats[source].totalRating / sourceStats[source].count).toFixed(1)
            }))
            .sort((a, b) => b.average - a.average)
            .slice(0, 3); // Păstrăm doar top 3
    }, [books]);
 
    // ==========================================
    // 3. Date false pentru graficul de Luni
    // ==========================================
    // Deoarece obiectele tale de Mock Data nu au o dată de adăugare reală,
    // vom crea un grafic vizual fix pentru acest exemplu, ca în design.
    // ==========================================
    // 3. Calcularea dinamică pentru Graficul pe Luni
    // ==========================================
    const monthlyData = useMemo(() => {
        // Pregătim contoarele pentru cele 6 luni pe care le afișăm
        const counts = { 'Jan': 0, 'Feb': 0, 'Mar': 0, 'Apr': 0, 'May': 0, 'Jun': 0 };
 
        books.forEach((book, index) => {
            let m = book.month; // Încercăm să luăm luna din carte
 
            // Dacă este o carte originală din mock data (nu are lună setată),
            // le distribuim vizual exact ca în designul tău inițial din Figma:
            Eif (!m) {
                if (index < 2) m = 'Jan';       // Primele 2 cărți în Ianuarie
                else if (index < 5) m = 'Feb';  // Următoarele 3 în Februarie
                else Em = 'Mar';                 // Restul în Martie
            }
 
            // Adăugăm +1 la luna respectivă
            Eif (counts[m] !== undefined) {
                counts[m] += 1;
            }
        });
 
        // Transformăm formatul ca să îl înțeleagă Recharts
        return Object.keys(counts).map(key => ({
            name: key,
            books: counts[key]
        }));
    }, [books]); // Se recalculează de fiecare dată când se modifică `books`
 
    return (
        <div className="insights-container">
            <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center', marginBottom: '2rem' }}>
                <h1 className="insights-title" style={{ marginBottom: 0 }}>Reading Insights</h1>
 
                {/* BUTONUL MAGIC */}
                <button
                    onClick={toggleAutoGenerate}
                    style={{
                        backgroundColor: isGenerating ? '#e74c3c' : 'var(--accent-orange)',
                        color: 'white',
                        border: 'none',
                        padding: '0.8rem 1.5rem',
                        borderRadius: '4px',
                        fontWeight: 'bold',
                        cursor: 'pointer',
                        transition: 'background-color 0.3s ease'
                    }}
                >
                    {isGenerating ? 'Stop' : 'Auto-Generate Data'}
                </button>
            </div>
 
            <div className="charts-grid">
                {/* Cardul 1: Bar Chart */}
                <div className="chart-card">
                    <h3>Books Scraped & Read by Month</h3>
                    <div className="chart-wrapper">
                        <ResponsiveContainer width="100%" height="100%">
                            <BarChart data={monthlyData}>
                                <CartesianGrid strokeDasharray="3 3" vertical={false} />
                                <XAxis dataKey="name" axisLine={false} tickLine={false} />
                                <YAxis axisLine={false} tickLine={false} />
                                <RechartsTooltip cursor={{fill: 'transparent'}}/>
                                <Bar dataKey="books" fill="#2C4A3E" radius={[4, 4, 0, 0]} />
                            </BarChart>
                        </ResponsiveContainer>
                    </div>
                </div>
 
                {/* Cardul 2: Donut Chart (Sincronizat cu RAM) */}
                <div className="chart-card">
                    <h3>Library by Genre</h3>
                    <div className="chart-wrapper">
                        <div className="donut-center-text">{books.length}</div>
                        <ResponsiveContainer width="100%" height="100%">
                            <PieChart>
                                <Pie
                                    data={genreData}
                                    innerRadius={60}
                                    outerRadius={100}
                                    paddingAngle={2}
                                    dataKey="value"
                                >
                                    {genreData.map((entry, index) => (
                                        <Cell key={`cell-${index}`} fill={COLORS[index % COLORS.length]} />
                                    ))}
                                </Pie>
                                <RechartsTooltip />
                            </PieChart>
                        </ResponsiveContainer>
                    </div>
                </div>
            </div>
 
            <div className="top-sources-card">
                <h3>Top Rated Sources</h3>
                <div className="sources-list">
                    {topSources.map((source, index) => (
                        <div key={index} className="source-row">
                            <div className="source-info">
                                <span className="source-rank">{index + 1}</span>
                                <span className="source-name">{source.name}</span>
                            </div>
                            <div className="source-score">
                                <span className="rating-stars">{"★".repeat(Math.round(source.average))}</span>
                                <span className="numeric-score">{source.average}</span>
                            </div>
                        </div>
                    ))}
                </div>
            </div>
        </div>
    );
}
